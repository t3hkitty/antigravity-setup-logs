import { useState, useEffect } from 'react';

/**
 * Custom hook for Sticky Settings with explicit localStorage keys,
 * fallback defaults, and instant change-listeners [cite: 615].
 */
export function useStickySetting<T>(key: string, defaultValue: T): [T, (value: T) => void] {
  const [value, setValue] = useState<T>(() => {
    try {
      const stored = window.localStorage.getItem(key);
      return stored !== null ? JSON.parse(stored) : defaultValue;
    } catch (error) {
      console.warn(`Error reading localStorage key "${key}":`, error);
      return defaultValue;
    }
  });

  useEffect(() => {
    try {
      window.localStorage.setItem(key, JSON.stringify(value));
    } catch (error) {
      console.warn(`Error setting localStorage key "${key}":`, error);
    }
    
    // Dispatch custom event for cross-tab instant sync
    window.dispatchEvent(new Event('anymd_harvester_state_change'));
  }, [key, value]);

  return [value, setValue];
}

import os
import re
import shutil
import json

# Configuration
INCOMING_DIR = os.path.expanduser("~/.antigravity/incoming_ideas/") # [cite: 664]
PLUGIN_QUEUE_DIR = os.path.expanduser("~/.antigravity/plugin_scaffold_queue/")
THRESHOLD_CONFIG = os.path.expanduser("~/.anymd_harvester_config.json")

# Regex Patterns for Implicit Code/Blueprint Detection [cite: 613, 1038]
REGEX_PATTERNS = {
    "low": [r"```(javascript|typescript|js|ts|python|py|bash|sh|json|yaml|yml)"],
    "medium": [r"(?i)blueprint", r"```\w+", r"(?i)plugin bundle"],
    "high": [r"(?i)const\s+", r"(?i)function\s+", r"(?i)import\s+.*from", r"(?i)export\s+"]
}

def load_sticky_settings():
    if os.path.exists(THRESHOLD_CONFIG):
        with open(THRESHOLD_CONFIG, "r") as f:
            return json.load(f)
    return {"anymd_harvester_auto_route": True, "anymd_harvester_regex_threshold": "medium"}

def analyze_and_route():
    settings = load_sticky_settings()
    if not settings.get("anymd_harvester_auto_route", True):
        print("[Harvester] Auto-routing disabled via sticky settings.")
        return

    threshold = settings.get("anymd_harvester_regex_threshold", "medium")
    active_patterns = REGEX_PATTERNS.get(threshold, REGEX_PATTERNS["medium"])
    
    os.makedirs(PLUGIN_QUEUE_DIR, exist_ok=True)
    
    for filename in os.listdir(INCOMING_DIR):
        filepath = os.path.join(INCOMING_DIR, filename)
        if not os.path.isfile(filepath):
            continue
            
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
            
        # Implicitly detect code language, text blocks, or blueprint keywords
        if any(re.search(pattern, content) for pattern in active_patterns):
            print(f"[Harvester] Implicit blueprint/code detected in {filename}. Routing to Plugin Queue.")
            shutil.move(filepath, os.path.join(PLUGIN_QUEUE_DIR, filename))

if __name__ == "__main__":
    analyze_and_route()
